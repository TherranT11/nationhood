/** * Parties Page * Displays all political parties/factions in the nation */ // Initialize page initPage('parties', async (state) => { const { nation, faction } = state; if (!nation) { document.getElementById('content-area').innerHTML = `
🏛️

No Nation Selected

Your party is not assigned to a nation yet.

`; return; } await renderPartiesPage(nation, faction); }); /** * Render the parties page content */ async function renderPartiesPage(nation, faction) { // Terminology based on government type const isAutocracy = nation.government_type === 'Autocracy'; const termPlural = isAutocracy ? 'Factions' : 'Parties'; const termSingular = isAutocracy ? 'Faction' : 'Party'; const headerText = isAutocracy ? 'Internal Factions' : 'Political Parties'; // Show loading in content area document.getElementById('content-area').innerHTML = `
${headerText} in ${nation.name}
Loading ${termPlural.toLowerCase()}...
`; try { // Get all parties in the nation const { data: parties, error } = await _supabase .from('factions') .select('*') .eq('nation_id', nation.id) .eq('faction_type', 'party') .order('approval_rating', { ascending: false }); if (error) throw error; if (!parties || parties.length === 0) { document.getElementById('content-area').innerHTML = `
${headerText} in ${nation.name}
No ${termPlural.toLowerCase()} in this nation yet.
`; return; } // Get the most recent election to determine seats (skip for autocracies) let election = null; if (!isAutocracy) { const { data: electionData } = await _supabase .from('elections') .select('*') .eq('nation_id', nation.id) .eq('status', 'completed') .order('election_tick', { ascending: false }) .limit(1) .maybeSingle(); election = electionData; } // Get active coalition to determine government/opposition status (skip for autocracies) let coalition = null; if (!isAutocracy) { const { data: coalitionData } = await _supabase .from('active_coalitions') .select('party_ids, lead_party_id') .eq('nation_id', nation.id) .order('formed_at', { ascending: false }) .limit(1) .maybeSingle(); coalition = coalitionData; } const coalitionPartyIds = coalition?.party_ids || []; const leadPartyId = coalition?.lead_party_id; const totalSeats = 120; const majorityThreshold = 61; // Build party cards const partiesHTML = parties.map(party => { // Get seats - for autocracies, use faction's seats directly let seats = 0; if (isAutocracy) { seats = party.seats || 0; } else if (election && election.results && election.results.seats) { const partyResult = election.results.seats.find(s => s.party_id === party.id); seats = partyResult?.seats || 0; } // Determine status based on coalition membership (different for autocracies) let status = 'opposition'; let statusText = 'Opposition'; if (isAutocracy) { // In autocracies, all factions are part of the regime status = 'government'; statusText = 'Regime'; } else if (coalitionPartyIds.includes(party.id)) { if (party.id === leadPartyId) { status = 'government'; statusText = 'Lead Party'; } else { status = 'coalition'; statusText = 'Coalition'; } } else if (seats >= majorityThreshold) { status = 'government'; statusText = 'Government'; } // Approval color const approvalClass = party.approval_rating >= 50 ? 'good' : party.approval_rating <= 30 ? 'bad' : ''; const isYourParty = party.id === faction?.id; return `
${party.faction_name} ${party.is_npc ? 'AI' : ''} ${isYourParty ? `★ YOUR ${termSingular.toUpperCase()}` : ''}
${party.abbreviation || '—'}
${statusText}
${party.ideology_value_1 ? `${party.ideology_value_1}` : ''} ${party.ideology_value_2 ? `${party.ideology_value_2}` : ''} ${isAutocracy && !party.ideology_value_1 ? 'Regime Faction' : ''}
Approval
${party.approval_rating || 0}%
Seats
${seats} / ${totalSeats}
Action Points
${party.action_points || 0}
${termSingular} Funds
$${(party.party_funds || 0).toLocaleString()}
`; }).join(''); // Count government vs opposition (for democracies) const governmentCount = isAutocracy ? parties.length : parties.filter(p => coalitionPartyIds.includes(p.id)).length; const oppositionCount = isAutocracy ? 0 : parties.length - governmentCount; // Render full page document.getElementById('content-area').innerHTML = `
${headerText} in ${nation.name}
Total ${termPlural}
${parties.length}
${isAutocracy ? 'In Regime' : 'Government'}
${governmentCount}
${isAutocracy ? 'Max Factions' : 'Opposition'}
${isAutocracy ? (nation.max_parties || 4) : oppositionCount}
Total Seats
${totalSeats}
${partiesHTML}
`; } catch (error) { console.error('Error loading parties:', error); document.getElementById('content-area').innerHTML = `
${headerText} in ${nation.name}
Error loading ${termPlural.toLowerCase()}. Please refresh the page.
`; } }
